Verified finite-difference tools for idealised one-dimensional slab cooling.

The package separates input contracts, numerical integration, independent analytical references and error metrics. Temperatures use degrees Celsius, positions metres and times seconds. Analytical agreement verifies the stated mathematical model; it does not constitute experimental physical validation.

 61    ----------
 62    intervals : int

 64    dt_s, end_time_s : float

 66        Snapshot times and the endpoint can shorten individual steps.
 67    method : {'ftcs', 'backward_euler'}

 69    profile : {'uniform', 'sine'}

105
106        Parameters

108        plate : PlateConfig, optional

110            uses ``min(dt_s, end_time_s)`` before snapshot splitting, making

112

116            If any control is invalid, the grid is unrepresentable, or the
117            normal FTCS diffusion number exceeds 0.5.
118        """

120                or not isinstance(self.intervals, Integral)
121                or self.intervals < 4):
122            raise ValueError("intervals must be an integer >= 4")
123        _finite_real("dt_s", self.dt_s, positive=True)
124        _finite_real("end_time_s", self.end_time_s, positive=True)
125        _finite_real("amplitude_c", self.amplitude_c)
126        _finite_real("perturbation_c", self.perturbation_c)
127        if self.method not in ("ftcs", "backward_euler"):
128            raise ValueError("method must be 'ftcs' or 'backward_euler'")
129        if self.profile not in ("uniform", "sine"):
130            raise ValueError("profile must be 'uniform' or 'sine'")
131        if self.implementation not in ("vectorized", "loop"):
132            raise ValueError("implementation must be 'vectorized' or 'loop'")
133        if not isinstance(self.allow_unstable, bool):
134            raise ValueError("allow_unstable must be a bool")
135        if plate is not None:

137            dx = plate.length_m / self.intervals
138            try:

140                diffusion_number = rate * min(self.dt_s, self.end_time_s)

 71def solve(plate: PlateConfig, settings: SolverConfig,
 72          snapshot_times: Sequence[float] | None = None) -> SimulationResult:
 73    """Integrate the slab heat equation with fixed equal face temperatures.
 74
 75    Parameters
 76    ----------
 77    plate : PlateConfig
 78        Physical inputs, with thickness in metres and diffusivity in m²/s.
 79    settings : SolverConfig
 80        Valid grid, integration and initial-condition controls.
 81    snapshot_times : sequence of float, optional
 82        Exact times in seconds at which to retain the full field, within
 83        ``[0, end_time_s]``. Times are sorted and deduplicated. The default
 84        retains the initial and final states without inserting extra internal
 85        events. An empty sequence retains no snapshots. The final field is
 86        always returned separately.
 87
 88    Returns
 89    -------
 90    SimulationResult
 91        Fields at the requested times plus a maximum-temperature history for
 92        every time step. Array dimensions are documented in the result class.
 93
 94    Raises
 95    ------
 96    ValueError
 97        For invalid inputs, snapshot times or an unstable normal FTCS setting.
 98    FloatingPointError
 99        If a diagnostic diverges to non-finite temperatures or a requested
100        step cannot advance floating-point time. Unstable finite temperatures
101        are returned without clipping.
102
103    Notes
104    -----
105    Both solvers use O(N) working field storage; the scalar maximum history
106    uses O(M), and requested snapshots use O(SN). FTCS updates every interior
107    point from the old state. Backward Euler solves the tridiagonal system
108    with known boundary contributions added to its right-hand side.
109
110    Examples
111    --------
112    >>> result = solve(PlateConfig(), SolverConfig(end_time_s=0.01))
113    >>> result.final_temperature_c.shape
114    (81,)
115    """
116    if not isinstance(plate, PlateConfig) or not isinstance(settings, SolverConfig):
117        raise ValueError("plate and settings must be PlateConfig and SolverConfig")
118    settings.validate(plate)
119    requested = _snapshot_schedule(snapshot_times, settings.end_time_s)
120    n = int(settings.intervals)
121    x = np.linspace(0.0, plate.length_m, n + 1)
122    dx = plate.length_m / n
123    rate = plate.diffusivity_m2_s / dx**2
124    if settings.profile == "uniform":
125        field = np.full(n + 1, plate.initial_temp_c, dtype=float)
126    else:
127        field = (plate.boundary_temp_c
128                 + settings.amplitude_c * np.sin(np.pi * x / plate.length_m))
129    if settings.perturbation_c:
130        field[1:-1] += settings.perturbation_c * (-1.0)**np.arange(1, n)
131    field[[0, -1]] = plate.boundary_temp_c
132    if not np.all(np.isfinite(field)):
133        raise ValueError("initial temperature calculation produced non-finite values")
134
135    times = [0.0]
136    maxima = [float(np.max(field))]
137    snapshots: list[NDArray[np.float64]] = []
138    snapshot_index = 0
139    if len(requested) and requested[0] == 0.0:
140        snapshots.append(field.copy())
141        snapshot_index = 1
142
143    t = 0.0
144    max_dt = 0.0
145    segment_start = 0.0
146    segment_step = 1
147    # Cache only the most recent step's matrix; a shortened snapshot step
148    # changes r, so it must not reuse an incompatible matrix.
149    previous_r = None
150    banded = None
151    while t < settings.end_time_s:
152        stop = settings.end_time_s
153        if snapshot_index < len(requested):
154            stop = float(requested[snapshot_index])
155        # Anchor each segment to its event start rather than repeatedly adding
156        # dt. This avoids cumulative drift and spurious microscopic end steps.
157        candidate = segment_start + segment_step * settings.dt_s
158        tolerance = 8 * np.finfo(float).eps * max(abs(stop), abs(candidate))
159        if candidate >= stop - tolerance:
160            next_t = stop
161        else:
162            next_t = candidate
163        dt = next_t - t
164        if dt <= 0 or not np.isfinite(dt):
165            raise FloatingPointError("time step cannot advance floating-point time")
166        r = rate * dt
167        with np.errstate(over="ignore", invalid="ignore"):
168            if settings.method == "ftcs":
169                updated = field.copy()
170                if settings.implementation == "vectorized":
171                    updated[1:-1] = (field[1:-1] + r
172                                     * (field[:-2] - 2 * field[1:-1] + field[2:]))
173                else:
174                    for i in range(1, n):
175                        updated[i] = field[i] + r * (field[i-1] - 2*field[i] + field[i+1])
176            else:
177                if previous_r != r:
178                    banded = np.zeros((3, n - 1))
179                    banded[0, 1:] = -r
180                    banded[1, :] = 1 + 2*r
181                    banded[2, :-1] = -r
182                    previous_r = r
183                rhs = field[1:-1].copy()
184                rhs[0] += r * plate.boundary_temp_c
185                rhs[-1] += r * plate.boundary_temp_c
186                updated = field.copy()
187                updated[1:-1] = solve_banded((1, 1), banded, rhs, check_finite=False)
188        if not np.all(np.isfinite(updated)):
189            raise FloatingPointError("non-finite temperature: diagnostic integration diverged")
190        field = updated
191        t = next_t
192        times.append(t)
193        maxima.append(float(np.max(field)))
194        max_dt = max(max_dt, dt)
195        if snapshot_index < len(requested) and t == requested[snapshot_index]:
196            snapshots.append(field.copy())
197            snapshot_index += 1
198        if t == stop:
199            segment_start = t
200            segment_step = 1
201        else:
202            segment_step += 1
203
204    stored = np.stack(snapshots) if snapshots else np.empty((0, n + 1), dtype=float)
205    return SimulationResult(
206        x_m=x, times_s=np.asarray(times), max_temperature_c=np.asarray(maxima),
207        temperature_snapshots_c=stored, snapshot_times_s=requested,
208        final_temperature_c=field.copy(), steps=len(times)-1, dt_max_s=max_dt,
209        diffusion_number=rate*max_dt, method=settings.method,
210    )

 62    """Evaluate a converged odd-sine series for uniform initial slab cooling.
 63
 64    Parameters
 65    ----------
 66    x_m : array_like
 67        Positions in metres, within ``[0, length_m]``.
 68    time_s : float
 69        Nonnegative time, in seconds. At exactly zero, return the prescribed
 70        uniform interior initial temperature and the imposed face values.
 71    plate : PlateConfig
 72        Physical inputs; the full thickness is used in every eigenvalue.
 73    terms : int
 74        Positive starting number of odd Fourier terms. It is doubled if an
 75        absolute tail bound is above 1e-11 degrees Celsius. At most 131072
 76        terms are used; times too close to zero can exceed this limit.
 77
 78    Returns
 79    -------
 80    ndarray
 81        Temperature in degrees Celsius, with the same shape as ``x_m``.
 82
 83    Raises
 84    ------
 85    ValueError
 86        For invalid inputs or if a positive evaluation time is too close to
 87        zero to satisfy the series truncation bound within the term limit.
 88
 89    Notes
 90    -----
 91    For ``a = alpha*(pi/L)**2*t`` and first omitted odd index ``q=2*K+1``,
 92    the absolute omitted series is bounded by
 93    ``4*abs(delta_T)/pi * (exp(-a*q*q)/q + exp1(a*q*q)/4)``.
 94    This follows from a decreasing-term integral bound and does not depend
 95    on fortuitous cancellation at particular spatial positions. The check
 96    bounds series truncation, not machine round-off or physical model error.
 97    """
 98    x = _coordinates(x_m, time_s, plate)
 99    if isinstance(terms, bool) or not isinstance(terms, Integral) or terms < 1:
100        raise ValueError("terms must be a positive integer")
101    if terms > 131072:
102        raise ValueError("terms must not exceed 131072")
103    boundary = (x == 0) | (x == plate.length_m)
104    if time_s == 0:
105        return np.where(boundary, plate.boundary_temp_c, plate.initial_temp_c)
106    delta = plate.initial_temp_c - plate.boundary_temp_c
107    if delta == 0:
108        return np.full_like(x, plate.boundary_temp_c)
109    a = plate.diffusivity_m2_s * (np.pi/plate.length_m)**2 * time_s
110    k = int(terms)
111    while True:
112        q = 2*k + 1
113        tail = 4*abs(delta)/np.pi * (np.exp(-a*q*q)/q + exp1(a*q*q)/4)
114        if tail <= 1e-11:
115            break
116        if k >= 131072:
117            raise ValueError("positive time is too close to zero for the series convergence limit")
118        k = min(2*k, 131072)
119    # Chunking bounds temporary storage even when an early time needs many
120    # modes; there is no dense mode-by-position history to retain.
121    total = np.zeros_like(x)
122    for start in range(0, k, 512):
123        odd = 2*np.arange(start, min(start+512, k), dtype=float) + 1
124        coefficients = np.exp(-a*odd**2)/odd
125        values = np.sin(odd[:, None] * np.pi * x.reshape(1, -1)/plate.length_m)
126        total += (coefficients @ values).reshape(x.shape)
127    result = plate.boundary_temp_c + delta*4/np.pi*total
128    return np.where(boundary, plate.boundary_temp_c, result)

42                            target_temp_c: float) -> float | None:
43    """Find the first at-or-below-target sample and interpolate its crossing.
44
45    Parameters
46    ----------
47    times_s : array_like, shape (M,)
48        Finite, nonnegative, strictly increasing sample times, in seconds.
49    max_temperature_c : array_like, shape (M,)
50        Finite sampled spatial maximum temperatures, in degrees Celsius.
51        Monotonic cooling is not assumed; the first downward crossing is used.
52    target_temp_c : float

54

56    -------
57    float or None

59        samples. Return the first time if the initial value is already at or
60        below target; return ``None`` if the target was never reached.
61
62    Raises
63    ------
64    ValueError
65        For unequal or invalid histories, non-increasing/negative times or a
66        non-finite threshold.
67    """
68    _finite_real("target_temp_c", target_temp_c)
69    times = np.asarray(times_s, dtype=float)
70    temperatures = np.asarray(max_temperature_c, dtype=float)
71    if (times.ndim != 1 or temperatures.ndim != 1 or times.shape != temperatures.shape
72            or times.size == 0):
73        raise ValueError("histories must be nonempty one-dimensional arrays with identical shapes")
74    if not np.all(np.isfinite(times)) or not np.all(np.isfinite(temperatures)):
75        raise ValueError("histories must contain only finite values")
76    if np.any(times < 0) or np.any(np.diff(times) <= 0):
77        raise ValueError("times must be nonnegative and strictly increasing")
78    hits = np.flatnonzero(temperatures <= target_temp_c)
79    if not len(hits):
80        return None
81    index = int(hits[0])
82    if index == 0:
83        return float(times[0])
84    fraction = ((temperatures[index-1] - target_temp_c)
85                / (temperatures[index-1] - temperatures[index]))
86    return float(times[index-1] + fraction*(times[index]-times[index-1]))
